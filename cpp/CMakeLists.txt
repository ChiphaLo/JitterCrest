cmake_minimum_required(VERSION 3.22)
project(speedtest_cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SPEEDTEST_SANITIZER "" CACHE STRING "address-undefined, thread, or empty")
option(SPEEDTEST_LTO "Enable comparable release link-time optimization" OFF)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(OpenSSL 3 REQUIRED)
find_package(Boost 1.83 REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
find_package(CURL 7.85 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(LibDataChannel 0.24.6 EXACT CONFIG REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/prepare_boost.py"
  --include "${Boost_INCLUDE_DIRS}" --output "${CMAKE_CURRENT_BINARY_DIR}/boost-overlay"
  COMMAND_ERROR_IS_FATAL ANY)
option(SPEEDTEST_WEB_UI "Embed the shared frontend (OFF builds an agent)" ON)
add_library(speedtest_core src/core.cpp src/control.cpp src/sdp.cpp src/config.cpp src/limits.cpp src/progress.cpp src/rtc_validation.cpp)
target_include_directories(speedtest_core BEFORE PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/boost-overlay" include)
target_link_libraries(speedtest_core PUBLIC nlohmann_json::nlohmann_json Boost::system OpenSSL::Crypto Threads::Threads)
target_compile_definitions(speedtest_core PUBLIC BOOST_ASIO_HAS_CO_AWAIT BOOST_ASIO_HAS_STD_COROUTINE)
target_compile_options(speedtest_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
if(SPEEDTEST_SANITIZER STREQUAL "address-undefined")
  set(SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(SPEEDTEST_SANITIZER STREQUAL "thread")
  set(SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
elseif(NOT SPEEDTEST_SANITIZER STREQUAL "")
  message(FATAL_ERROR "Unsupported sanitizer configuration")
endif()
if(SANITIZER_FLAGS)
  target_compile_options(speedtest_core PUBLIC ${SANITIZER_FLAGS})
  target_link_options(speedtest_core PUBLIC ${SANITIZER_FLAGS})
endif()
if(SPEEDTEST_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT IPO_OK OUTPUT IPO_ERROR)
  if(NOT IPO_OK)
    message(FATAL_ERROR "LTO unavailable: ${IPO_ERROR}")
  endif()
  set_property(TARGET speedtest_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
include(CTest)
add_executable(speedtest-core-fixture tests/core_fixture.cpp)
target_link_libraries(speedtest-core-fixture PRIVATE speedtest_core)
add_test(NAME protocol_statistics_control_sdp COMMAND speedtest-core-fixture)
option(SPEEDTEST_FUZZ "Build the bounded core libFuzzer harness (Clang ASan/UBSan)" OFF)
if(SPEEDTEST_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT SPEEDTEST_SANITIZER STREQUAL "address-undefined")
    message(FATAL_ERROR "Fuzzing requires Clang and the separate address-undefined build")
  endif()
  target_compile_options(speedtest_core PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(speedtest-core-fuzz tests/fuzz_core.cpp)
  target_link_libraries(speedtest-core-fuzz PRIVATE speedtest_core)
  target_compile_options(speedtest-core-fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(speedtest-core-fuzz PRIVATE -fsanitize=fuzzer)
endif()

add_executable(speedtest-config-fixture tests/config_fixture.cpp)
target_link_libraries(speedtest-config-fixture PRIVATE speedtest_core)
add_test(NAME configuration_bounds COMMAND speedtest-config-fixture)
add_executable(speedtest-limits-test tests/limits_test.cpp)
target_link_libraries(speedtest-limits-test PRIVATE speedtest_core)
add_test(NAME raii_limits_concurrency COMMAND speedtest-limits-test)
add_executable(speedtest-progress-test tests/progress_test.cpp)
target_link_libraries(speedtest-progress-test PRIVATE speedtest_core)
add_test(NAME upload_progress_lifecycle COMMAND speedtest-progress-test)
add_executable(speedtest-rtc-validation-test tests/rtc_validation_test.cpp)
target_link_libraries(speedtest-rtc-validation-test PRIVATE speedtest_core)
add_test(NAME rtc_upload_complete_reasons COMMAND speedtest-rtc-validation-test)
add_executable(speedtest-executor-refcount-test tests/executor_refcount_test.cpp)
target_link_libraries(speedtest-executor-refcount-test PRIVATE speedtest_core)
add_test(NAME asio_executor_refcount_release COMMAND speedtest-executor-refcount-test)

set(SHARED_ASSETS "${CMAKE_CURRENT_SOURCE_DIR}/../public" CACHE PATH "Unmodified shared frontend")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/assets.cpp"
  COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_assets.py"
    --assets "${SHARED_ASSETS}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated/assets.cpp"
    --web-ui "${SPEEDTEST_WEB_UI}"
  DEPENDS cmake/embed_assets.py "${SHARED_ASSETS}/index.html" "${SHARED_ASSETS}/favicon.svg"
  VERBATIM)
add_library(speedtest_http src/client_info.cpp src/http.cpp src/rtc.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/assets.cpp")
target_link_libraries(speedtest_http PUBLIC speedtest_core CURL::libcurl LibDataChannel::LibDataChannel)
# Boost 1.83's Clang detection predates Clang 18's C++20 coroutine macros.
target_compile_definitions(speedtest_http PUBLIC BOOST_ASIO_HAS_CO_AWAIT BOOST_ASIO_HAS_STD_COROUTINE)
target_compile_options(speedtest_http PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
add_executable(speedtest-cpp src/main.cpp)
target_link_libraries(speedtest-cpp PRIVATE speedtest_http)
target_compile_options(speedtest-cpp PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
if(SPEEDTEST_LTO)
  set_property(TARGET speedtest_http speedtest-cpp PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
add_executable(speedtest-client-info-test tests/client_info_test.cpp)
target_link_libraries(speedtest-client-info-test PRIVATE speedtest_http)
add_test(NAME bounded_client_info_provider COMMAND speedtest-client-info-test)
